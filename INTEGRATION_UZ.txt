async def language_command(update, context):
    await update.effective_message.reply_text(
        language_selection_text(),
        parse_mode="HTML",
        reply_markup=language_keyboard()
    )

async def language_callback(update, context):
    query = update.callback_query
    await query.answer()
    data = query.data or ""

    if data == "language":
        await query.edit_message_text(
            language_selection_text(),
            parse_mode="HTML",
            reply_markup=language_keyboard()
        )
        return

    if data.startswith("setlang:"):
        lang = data.split(":", 1)[1]
        if lang not in ("uz", "ru", "en"):
            lang = "uz"

        chat = query.message.chat if query.message else None
        user = query.from_user

        # Guruhdagi tanlov guruh tilini o‘zgartiradi.
        # Shaxsiy chatdagi tanlov faqat shu foydalanuvchiga tegishli.
        if chat and chat.type in ("group", "supergroup"):
            set_language("chat", chat.id, lang)
        elif user:
            set_language("user", user.id, lang)

        await query.edit_message_text(
            tr(lang, "language_saved") +
            "\n\n" + tr(lang, "start_choose"),
            parse_mode="HTML"
        )
